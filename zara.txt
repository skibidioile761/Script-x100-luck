-- sounds (replace SoundId if you have favorite assets)
local function playTone(soundId, volume, speed)
	local s = Instance.new("Sound")
	s.SoundId = soundId
	s.Volume = volume or 0.35
	s.PlaybackSpeed = speed or 1
	s.Parent = SoundService
	s:Play()
	Debris:AddItem(s, 2)
end

-- small helper tween
local function tween(obj, t, props, style, dir)
	local tw = TweenService:Create(
		obj,
		TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	tw:Play()
	return tw
end

-- drag logic
local function makeDraggable(frame, handle)
	local dragging = false
	local dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	handle.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end)
end

-- state
local isOn = false
local effectRun = false
local particleCon
local orbPulseTw

local BOOST_MESSAGE = "Boost 100% luck for your next roll(100% Got Legendary or Mythical)"

local function makeParticle()
	local p = Instance.new("Frame")
	p.Size = UDim2.new(0, math.random(3, 7), 0, math.random(3, 7))
	p.Position = UDim2.new(0.5, math.random(-16, 16), 1, -20)
	p.BackgroundColor3 = Color3.fromRGB(math.random(130, 190), 255, math.random(170, 235))
	p.BackgroundTransparency = 0.05
	p.BorderSizePixel = 0
	p.Parent = status
	Instance.new("UICorner", p).CornerRadius = UDim.new(1, 0)

	local dx = math.random(-60, 60)
	local dur = math.random(85, 140) / 100
	local tw = tween(p, dur, {
		Position = UDim2.new(0.5, dx, 0, -22),
		BackgroundTransparency = 1
	}, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

	tw.Completed:Connect(function()
		if p then p:Destroy() end
	end)
end

local function startEffects()
	if effectRun then return end
	effectRun = true

	statusText.Text = "Boost Luck: ON"
	statusText.TextColor3 = Color3.fromRGB(146, 255, 120)
	msg.Text = BOOST_MESSAGE

	tween(orb, 0.22, {BackgroundTransparency = 0.05})
	tween(orbGlow, 0.22, {Transparency = 0.15})
	tween(mainStroke, 0.3, {Transparency = 0.35})

	if orbPulseTw then orbPulseTw:Cancel() end
	orbPulseTw = TweenService:Create(
		orb,
		TweenInfo.new(0.85, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
		{Size = UDim2.new(0, 62, 0, 62), Position = UDim2.new(0.5, -31, 0, 7)}
	)
	orbPulseTw:Play()

	particleCon = RunService.RenderStepped:Connect(function(dt)
		if math.random() < (dt * 7.5) then
			makeParticle()
		end
	end)
end

local function stopEffects()
	effectRun = false

	statusText.Text = "Boost Luck: OFF"
	statusText.TextColor3 = Color3.fromRGB(140, 147, 160)
	msg.Text = ""

	tween(orb, 0.25, {BackgroundTransparency = 0.25, Size = UDim2.new(0, 56, 0, 56), Position = UDim2.new(0.5, -28, 0, 10)})
	tween(orbGlow, 0.25, {Transparency = 0.5})
	tween(mainStroke, 0.25, {Transparency = 0.65})

	if orbPulseTw then
		orbPulseTw:Cancel()
		orbPulseTw = nil
	end
	if particleCon then
		particleCon:Disconnect()
		particleCon = nil
	end
end

local function toggleBoost()
	isOn = not isOn
	if isOn then
		playTone(SND_ON_A, 0.35, 1.16)
		task.delay(0.03, function()
			playTone(SND_ON_B, 0.2, 1.34)
		end)

		tween(switch, 0.22, {BackgroundColor3 = Color3.fromRGB(0, 214, 166)})
		tween(knob, 0.22, {Position = UDim2.new(0, 27, 0.5, -12)}, Enum.EasingStyle.Back)
		startEffects()
	else
		playTone(SND_OFF, 0.33, 0.85)

		tween(switch, 0.22, {BackgroundColor3 = Color3.fromRGB(52, 58, 70)})
		tween(knob, 0.22, {Position = UDim2.new(0, 3, 0.5, -12)}, Enum.EasingStyle.Back)
		stopEffects()
	end
end

-- loading flow
task.spawn(function()
	local progress = 0
	local shimmerOffset = -70

	while progress < 100 do
		progress = math.min(100, progress + math.random(3, 8))
		percent.Text = tostring(progress) .. "%"
		tween(barFill, 0.12, {Size = UDim2.new(progress / 100, 0, 1, 0)}, Enum.EasingStyle.Quad)

		shimmerOffset = shimmerOffset + 18
		shimmer.Position = UDim2.new(0, shimmerOffset, 0, 0)
		if shimmerOffset > 380 then shimmerOffset = -70 end

		task.wait(0.08)
	end

	task.wait(0.2)
	playTone(SND_OPEN, 0.26, 1.06)

	local f1 = tween(loading, 0.42, {BackgroundTransparency = 1})
	tween(loadTitle, 0.42, {TextTransparency = 1})
	tween(loadSub, 0.42, {TextTransparency = 1})
	tween(percent, 0.42, {TextTransparency = 1})
	tween(barTrack, 0.42, {BackgroundTransparency = 1})
	tween(barFill, 0.42, {BackgroundTransparency = 1})

	f1.Completed:Connect(function()
		loading.Visible = false

		main.Visible = true
		main.Position = UDim2.new(0.5, -175, 0.12, 0)
		main.BackgroundTransparency = 1
		main.Size = UDim2.new(0, 334, 0, 250)

		tween(main, 0.32, {
			Position = UDim2.new(0.5, -175, 0.16, 0),
			BackgroundTransparency = 0.2,
			Size = UDim2.new(0, 350, 0, 270)
		}, Enum.EasingStyle.Back)
	end)
end)
